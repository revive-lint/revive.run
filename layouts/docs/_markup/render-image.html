{{- /* Overrides the lotusdocs hook, which pipes every image through resources.GetRemote: that makes the build depend on each remote host, and a slow or rate-limiting one fails it. Remote images are left to the browser; local ones keep the theme's behaviour. */ -}} {{- $remote := ne (urls.Parse .Destination).Scheme "" -}} {{- $resource := "" -}} {{- if not $remote -}} {{- $resource = resources.Get .Destination -}} {{- end -}} {{- define "_partials/inline/image.html" -}} {{- with .resource -}} {{- if eq .MediaType.SubType "svg" -}} {{- .Content | safeHTML -}} {{- else -}} {{- end -}} {{- else -}} {{- end -}} {{- end -}} {{- $image := dict "resource" $resource "text" .PlainText "destination" .Destination -}} {{ with .Title }} {{ else }} {{ partial "inline/image.html" $image }} {{ end }}
